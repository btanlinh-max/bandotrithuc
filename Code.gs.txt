/* Bản Đồ Tri Thức – lõi xử lý dùng chung cho Google Apps Script (máy chủ của cô) và bản dùng thử trên trình duyệt.
   Bộ chuyển (adapter) A cung cấp: all(bảng), put(bảng, bản ghi), del(bảng, id), secret(khóa[, giá trị]),
   sign(chuỗi), hash(chuỗi), putFile(tên, base64), getFile(mã).
   Hành động cần AI trả về {__ai: parts, json, then(text, err)}; nơi gọi tự chạy AI rồi gọi then. */

  function uid(p) { return p + Date.now().toString(36) + Math.random().toString(36).slice(2, 6); }
  function now() { return new Date().toISOString(); }
  function stars(s) { return s >= 9 ? 3 : s >= 7 ? 2 : 1; }
  function norm(v) {
    return String(v == null ? "" : v).toLowerCase().normalize("NFC").replace(/\s+/g, " ").replace(/[.。]+$/, "").trim();
  }
  function numOrNull(v) { var s = String(v).replace(/\s/g, "").replace(",", "."); return /^-?\d+(\.\d+)?$/.test(s) ? Number(s) : null; }
  function err(msg, code) { var e = new Error(msg); e.code = code || "loi"; return e; }
  function extractJSON(t) {
    t = String(t || "").replace(/```json|```/gi, "").trim();
    var a = t.search(/[\[{]/); if (a < 0) throw err("AI trả lời không đúng dạng, cô thử lại nhé.");
    var close = t[a] === "{" ? "}" : "]"; return JSON.parse(t.slice(a, t.lastIndexOf(close) + 1));
  }

    function del(t, id) { A.del(t, id); cache[t] = all(t).filter(function (x) { return x.id !== id; }); }
    function get(t, id) { return all(t).find(function (x) { return x.id === id; }); }

    function token(role, sid) { var exp = Date.now() + 60 * DAY, body = role + "." + (sid || "-") + "." + exp; return body + "." + A.sign(body); }
    function who(tok) {

      return ok.some(function (x) { var xn = numOrNull(x); return (n != null && xn != null) ? n === xn : norm(v) === x; });
    }

      });

    function cleanQ(q, i) {
      var t = ["mcq", "tf", "fill", "short", "photo"].indexOf(q.type) >= 0 ? q.type : "short";
      var o = { id: uid("q") + i, type: t, level: [1, 2, 3].indexOf(Number(q.level)) >= 0 ? Number(q.level) : 1, text: String(q.text || "").trim(), points: Number(q.points) === 2 ? 2 : 1, explain: String(q.explain || ""), ai: true };
      if (t === "mcq") { o.options = (q.options || []).map(String).slice(0, 4); o.answer = Math.max(0, Math.min(o.options.length - 1, Number(q.answer) || 0)); }
      else if (t === "tf") o.answer = q.answer === true || String(q.answer).toLowerCase() === "true" || q.answer === "đúng";
      else o.answer = String(q.answer == null ? "" : q.answer);

    }

    /* ---------- chấm ---------- */
    function finishGrade(sub, a, parsed) {
      var total = totalPoints(a), got = 0, pending = false;
      a.questions.forEach(function (q) {
        if (AUTO_TYPES[q.type]) { if (sub.auto[q.id]) got += Number(q.points) || 1; return; }
        var r = parsed && parsed.items && parsed.items.find(function (x) { return x.qid === q.id; });
        if (r) {
          var sc = r.score == null ? null : Math.max(0, Math.min(Number(q.points) || 1, Number(r.score)));
          sub.ai[q.id] = { score: sc, comment: String(r.comment || ""), note: String(r.note || "") };
          if (sc == null) pending = true; else got += sc;
        } else pending = true;
      });
      sub.draft10 = Math.round(got / total * 10);
      sub.aiOverall = parsed && parsed.overall ? String(parsed.overall) : "";
      sub.needCheck = pending;
      return sub;
    }

var TABLES = { settings: "Cài đặt", students: "Học sinh", lessons: "Bài học", asgs: "Bài giao", subs: "Bài nộp", log: "Nhật kí" };
var SS = SpreadsheetApp.getActiveSpreadsheet();

function sheet_(t) {
  var sh = SS.getSheetByName(TABLES[t]);
  if (!sh) {
    sh = SS.insertSheet(TABLES[t]);
    sh.getRange(1, 1, 1, 4).setValues([["id", "Cập nhật", "Tóm tắt", "Dữ liệu (không sửa tay)"]]).setFontWeight("bold");
    sh.setFrozenRows(1); sh.setColumnWidth(3, 320);
  }
  return sh;
}
function summary_(t, r) {
  try {
    if (t === "students") return r.name;
    if (t === "lessons") return (r.subject === "tv" ? "Tiếng Việt" : "Toán") + " – " + r.topic + " – " + r.title;
    if (t === "asgs") return r.title + " (" + r.questions.length + " câu, hạn " + (r.due || "—") + ")";
    if (t === "subs") return "Điểm " + (r.score10 == null ? "chưa chấm" : r.score10) + " – " + (r.status === "da_tra" ? "đã trả" : "chờ cô duyệt");
    if (t === "log") return r.type;
  } catch (e) {}
  return "";
}
function rowOf_(sh, id) {
  var n = sh.getLastRow(); if (n < 2) return -1;
  var ids = sh.getRange(2, 1, n - 1, 1).getValues();
  for (var i = 0; i < ids.length; i++) if (ids[i][0] === id) return i + 2;
  return -1;
}
var GasAdapter = {
  all: function (t) {
    var sh = sheet_(t), n = sh.getLastRow(); if (n < 2) return [];
    return sh.getRange(2, 4, n - 1, 1).getValues().map(function (r) { try { return JSON.parse(r[0]); } catch (e) { return null; } }).filter(Boolean);
  },
  put: function (t, r) {
    var sh = sheet_(t), row = rowOf_(sh, r.id), vals = [[r.id, new Date(), summary_(t, r), JSON.stringify(r)]];
    if (row < 0) sh.appendRow(vals[0]); else sh.getRange(row, 1, 1, 4).setValues(vals);
  },
  del: function (t, id) { var sh = sheet_(t), row = rowOf_(sh, id); if (row > 0) sh.deleteRow(row); },
  secret: function (k, v) { if (v === undefined) return PROPS.getProperty("bdtt_" + k) || ""; PROPS.setProperty("bdtt_" + k, v); },
  sign: function (s) {
    var key = PROPS.getProperty("bdtt_sign"); if (!key) { key = Utilities.getUuid() + Utilities.getUuid(); PROPS.setProperty("bdtt_sign", key); }
    return Utilities.base64EncodeWebSafe(Utilities.computeHmacSha256Signature(s, key)).replace(/[=.]/g, "").slice(0, 24);

function callGemini_(parts, json, key, model) {
  if (!key) throw new Error("Cô chưa nhập mã AI Gemini (vào Cài đặt).");
  var models = [model || "gemini-2.5-flash", "gemini-flash-latest", "gemini-2.0-flash"].filter(function (m, i, a) { return m && a.indexOf(m) === i; }), last;
  for (var i = 0; i < models.length; i++) {
    var res = UrlFetchApp.fetch("https://generativelanguage.googleapis.com/v1beta/models/" + models[i] + ":generateContent", {
      method: "post", contentType: "application/json", muteHttpExceptions: true, headers: { "x-goog-api-key": key },
      payload: JSON.stringify({ contents: [{ role: "user", parts: parts }], generationConfig: json ? { temperature: 0.5, responseMimeType: "application/json" } : { temperature: 0.5 } })
    });
    var code = res.getResponseCode(), body = res.getContentText();
    if (code === 404) { last = new Error("Không tìm thấy mô hình AI " + models[i]); continue; }
    if (code === 429) throw new Error("Hết lượt AI miễn phí trong lúc này. Cô chờ 1 phút rồi thử lại.");
    if (code >= 400) { var msg = ""; try { msg = JSON.parse(body).error.message; } catch (e) {} throw new Error("Mã AI chưa dùng được (Google báo " + code + ": " + msg + ")."); }
    var d = JSON.parse(body);
    return ((((d.candidates || [])[0] || {}).content || {}).parts || []).map(function (p) { return p.text || ""; }).join("").trim();
  }
  throw last || new Error("Không gọi được AI.");
}

var WRITE = { init: 1, studentLogin: 1, submit: 1, practiceDone: 1, saveSettings: 1, saveStudents: 1, saveLessons: 1, saveAsg: 1, delAsg: 1, review: 1, regrade: 1, aiGen: 1, aiToc: 1 };
function run_(action, payload, token) {
  var lock = LockService.getScriptLock(), locked = false;
  function lockIt() { if (WRITE[action]) { lock.waitLock(25000); locked = true; } }
  function unlock() { if (locked) { lock.releaseLock(); locked = false; } }
  try {
    lockIt();
    var core = Core.make(GasAdapter), r = core.handle(action, payload, token);
    while (r && r.__ai) {
      var key = core.aiKey(), model = core.model(); unlock();   // không giữ khóa trong lúc chờ AI
      var text = null, e = null;
      try { text = callGemini_(r.__ai, r.json, key, model); } catch (x) { e = x; }
      lockIt(); r = r.then(text, e);
    }
    return { ok: true, data: r };
  } catch (e) {
    return { ok: false, error: String(e && e.message || e), code: e && e.code || "loi" };
  } finally { unlock(); }
}
function doPost(e) {
  var body = {}; try { body = JSON.parse(e.postData.contents); } catch (x) {}
  return ContentService.createTextOutput(JSON.stringify(run_(body.action, body.payload, body.token))).setMimeType(ContentService.MimeType.JSON);
}
